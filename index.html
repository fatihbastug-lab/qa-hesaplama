function hesapla() { const simdi = new Date(); const baslangic = new Date(); baslangic.setHours(8, 30, 0); const bitis = new Date(); bitis.setHours(18, 45, 0); // Sabit Parametreler const netCalismaSuresi = 540; // 615dk toplam - 75dk mola const minHedefOranı = 0.65; // Minimum %65 hedef const hedefDk = netCalismaSuresi * minHedefOranı; // 351 dk let gecenBrutDk = Math.floor((simdi - baslangic) / 60000); gecenBrutDk = Math.max(0, Math.min(gecenBrutDk, 615)); // Veriler const nA = parseFloat(document.getElementById('adet').value) || 0; const nS = parseFloat(document.getElementById('sure').value) || 0; const eA = parseFloat(document.getElementById('ekAdet').value) || 0; const eS = parseFloat(document.getElementById('ekSure').value) || 0; const tDk = parseFloat(document.getElementById('toplanti').value) || 0; // Üretim Formülü (Sabit + Değişken + Ek İş + Toplantı) const uretim = (nA * nS) + (nA * 2) + (eA * (eS * 0.5)) + (tDk * 0.70); // Verimlilik Yüzdesi const toplamVerim = ((uretim / netCalismaSuresi) * 100).toFixed(1); // UI Güncelleme - Badge const vBadge = document.getElementById('verimSonuc'); vBadge.style.display = 'block'; vBadge.innerText = `%${toplamVerim} VERİMLİLİK`; // KRİTİK: %65 altı DANGER, %65 ve üzeri SUCCESS if (toplamVerim >= 65) { vBadge.className = 'res-badge success'; } else { vBadge.className = 'res-badge danger'; } // Analiz ve Tahminler const kalanHedefDk = Math.max(0, hedefDk - uretim); const kalanDk = Math.max(0, 615 - gecenBrutDk); const ortalamaSure = (nS + 2) || 10; // nS girilmemişse varsayılan 10dk const gerekenAdet = Math.ceil(kalanHedefDk / ortalamaSure); document.getElementById('istatistikler').style.display = 'grid'; document.getElementById('uretimDk').innerText = `${uretim.toFixed(0)} dk Üretilen`; // Anlık mola/kayıp hesabı (Orantısal mola düşülerek) let anlikMolaPayi = (gecenBrutDk / 615) * 75; let gecenNetDk = gecenBrutDk - anlikMolaPayi; let kayipDk = Math.max(0, gecenNetDk - uretim); document.getElementById('molaDk').innerText = `${kayipDk.toFixed(0)} dk Kayıp`; document.getElementById('analizDetay').style.display = 'block'; document.getElementById('analizDetay').innerHTML = ` • Durum: ${toplamVerim >= 65 ? 'Hedef Üstü' : 'Hedef Altı'}
• Minimum Hedef (%65): 351 dk
• Hedefe kalan üretim: ${kalanHedefDk.toFixed(0)} dk
• Gereken minimum ek işlem: ${gerekenAdet} adet `; document.getElementById('footerButtons').style.display = 'flex'; }
